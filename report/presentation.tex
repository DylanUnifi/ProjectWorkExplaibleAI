\begin{frame}{Context and Motivation}
    \textbf{The "Black Box" Problem}
    \vspace{0.2cm}
    \begin{itemize}
        \item Deep learning models (CNNs, ViTs) have opaque decision-making processes.
        \item \textbf{Post-Hoc XAI}: Explainers applied \textit{after} training (e.g., \textbf{SHAP} \cite{lundberg2017unified}, \textbf{LIME} \cite{ribeiro2016should}).
        \item \textbf{Intrinsic XAI}: Models interpretable by design (e.g., \textbf{ProtoPNet} \cite{chen2019this}).
    \end{itemize}
\end{frame}

\begin{frame}{LIME Ablation Study on CLE4EVR}
    \begin{itemize}
        \item \textbf{Hypothesis}: Does matching LIME's superpixels ($K$) to the number of objects improve explanations?
        \item \textbf{Result}: Object-aligned boundaries ($K=4$) unexpectedly double the \textit{Infidelity} compared to default ($K=50$). Linear surrogates fail to capture the non-linear reasoning of CNNs.
    \end{itemize}

\begin{frame}{Conclusion}
    \begin{block}{Key Takeaways}
        \begin{itemize}
            \item Relying strictly on "Localization Accuracy" is dangerous: it assumes networks reason using human-defined object boundaries.
            \item Intrinsic interpretability reveals that models learn broad spatial contexts, unlike what post-hoc methods suggest \cite{rudin2019stop}.
        \end{itemize}
    \end{block}
